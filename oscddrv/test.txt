# oscddrv/test.txt
# oscddrv 字符设备注册与自动节点创建测试

# 1. 编译模块（若需要更新 IDE 提示可改用 bear -- make）
make
# 预期：无错误，生成 oscddrv.ko 文件

# 2. 加载模块
sudo insmod oscddrv.ko
# 预期：无错误输出，模块插入内核

# 3. 查看设备节点
ls -l /dev/oscddrv
# 预期：输出类似 crw-rw-rw- 1 root root 238, 0 ... /dev/oscddrv，表示字符设备已创建

# 4. 查看内核日志（应包含设备号分配信息）
dmesg | tail -5
# 预期：包含 "oscddrv: allocated device number (major=238, minor=0)" 和 "module loaded successfully"

# 5. 确认模块在已加载列表中
lsmod | grep oscddrv
# 预期：有 oscddrv 条目，显示大小和使用计数（通常 0）

# 6. 卸载模块
sudo rmmod oscddrv
# 预期：无错误输出

# 7. 确认设备节点已自动删除
ls -l /dev/oscddrv
# 预期：ls: cannot access '/dev/oscddrv': No such file or directory

# 8. 再次查看内核日志，确认卸载成功
dmesg | tail -3
# 预期：包含 "oscddrv: module unloaded successfully"
