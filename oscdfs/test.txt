# oscdfs/test.txt
# 实验二核心功能测试（含 pwd 和 dir 新格式）
# 启动前请执行 ./oscdfs --init 确保磁盘干净

# 1. 初始状态（自动 root 登录，家目录 /home/root）
pwd
# /home/root
dir
# 只有 . 和 ..

# 2. 根目录
dir /
# 应有 . .. home
pwd
# /home/root（不变）

# 3. 查看 /home 下用户
dir /home
# . .. root oscd pyc guest

# 4. 切换用户 oscd
login oscd oscd
# logged in as 'oscd' (uid=1000, gid=1000)
pwd
# /home/oscd
dir
# . ..

# 5. oscd 创建、写入、读取、删除文件（完整生命周期）
create /home/oscd/notes.txt
# created '/home/oscd/notes.txt' (inode X)
dir
# notes.txt 出现，owner uid=1000
open /home/oscd/notes.txt rw
# opened as fd 0
write 0 testdata
# (8 bytes written)
close 0
open /home/oscd/notes.txt r
# opened as fd 0
read 0 100
# testdata
# (8 bytes read)
close 0
delete /home/oscd/notes.txt
# deleted
dir
# notes.txt 消失

# 6. root 创建文件并设置权限，测试跨用户访问
login root root
pwd
# /home/root
create /home/root/testfile
open /home/root/testfile rw
write 0 Hello
close 0
chmod /home/root/testfile 600
# chmod to 600

# 7. oscd 尝试读取（other 无权限）
login oscd oscd
open /home/root/testfile r
# 预期错误：permission denied

# 8. oscd 不能 chmod 非自己文件
chmod /home/root/testfile 644
# 预期错误：permission denied

# 9. root 修改文件所有者为 oscd，然后 oscd 可读
login root root
chown /home/root/testfile 1000
# chown uid=1000
login oscd oscd
open /home/root/testfile r
# opened as fd 0
close 0

# 10. oscd 无法删除该文件（目录 /home/root 属于 root 不可写）
delete /home/root/testfile
# 预期错误：permission denied

# 11. root 清理测试文件
login root root
delete /home/root/testfile
# deleted

# 12. cd 无参数回家目录，确认 pwd 更新
cd /
pwd
# /
dir
# 根目录内容
cd
pwd
# /home/root (当前 root)
dir
# /home/root 内容

# 13. 退出
exit
# Bye from oscdfs!
