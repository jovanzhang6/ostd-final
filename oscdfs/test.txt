# oscdfs/test.txt

# 1. 列出根目录
dir /
# 预期输出（表格）：inode 1 的 . 和 .. ，权限 000755，大小 4096

# 2. 浏览当前目录（默认根）
dir
# 同上

# 3. 切换到不存在目录
cd /nonexistent
# 预期错误：oscdfs: cd: path '/nonexistent' does not exist （注意无“未知命令”）

# 4. 切换至根目录
cd /
# 无输出，当前工作目录仍为根

# 5. 创建文件
create /hello.txt
# 预期输出：created '/hello.txt' (inode 2)

# 6. 再次创建同名文件
create /hello.txt
# 预期错误：oscdfs: oscdfs_create: '/hello.txt' already exists

# 7. 查看根目录，确认文件存在
dir /
# 预期输出表格包含：2  hello.txt  0  000644 0

# 8. 以读写模式打开文件
open /hello.txt rw
# 预期输出：opened '/hello.txt' as fd 0

# 9. 写入字符串
write 0 HelloWorld
# 预期输出：(10 bytes written)

# 10. 立即读取（偏移已到末尾）
read 0 5
# 预期输出：(0 bytes read)

# 11. 关闭文件
close 0
# 预期输出：closed fd 0

# 12. 重新以只读打开，验证持久化
open /hello.txt r
# 预期输出：opened '/hello.txt' as fd 0   （重用 fd 0）

# 13. 读取大量字节，验证全部内容
read 0 100
# 预期输出：HelloWorld （下一行显示 (10 bytes read)）

# 14. 关闭
close 0
# closed fd 0

# 15. 再次打开 rw 准备覆盖测试（重用 fd 0）
open /hello.txt rw
# 预期输出：opened '/hello.txt' as fd 0

# 16. 写入三个感叹号（覆盖开头3字节）
write 0 !!!
# (3 bytes written)

# 17. 关闭
close 0
# closed fd 0

# 18. 只读打开并读取
open /hello.txt r
# opened '/hello.txt' as fd 0
read 0 100
# 预期输出：!!!loWorld （下一行 (10 bytes read)）

# 19. 关闭
close 0
# closed fd 0

# 20. 删除文件
delete /hello.txt
# 预期输出：deleted '/hello.txt'

# 21. 再次删除
delete /hello.txt
# 预期错误：oscdfs: oscdfs_delete: '/hello.txt' not found

# 22. 确认目录中文件消失
dir /
# 只显示 . 和 .. ，无 hello.txt

# 23. 测试错误处理：打开不存在文件
open /no.txt
# 预期错误：oscdfs: oscdfs_open: '/no.txt' not found

# 24. 以只读打开后尝试写入
create /ro.txt
# 预期输出：created '/ro.txt' (inode 2)  （inode 2 已被回收）
open /ro.txt r
# opened '/ro.txt' as fd 0
write 0 test
# 预期错误：oscdfs: oscdfs_write: fd 0 is read-only
close 0
# closed fd 0
delete /ro.txt
# deleted '/ro.txt'

# 25. 关闭无效 fd
close 99
# 预期错误：oscdfs: oscdfs_close: invalid fd 99

# 26. 退出
exit
# Bye from oscdfs! 并返回shell
