# oscdsh/test.txt

# 用户
whoami
sudo -i
whoami
exit

# 准备
mkdir test

# 重定向
echo hello > test/a.txt
cat test/a.txt                 # hello
echo world >> test/a.txt
cat test/a.txt                 # hello\nworld
wc -c < test/a.txt             # 字节数（为12）
cat < test/a.txt > test/b.txt
cat test/b.txt                 # hello\nworld
> test/empty.txt
ls -l test/empty.txt           # 大小 0

# 管道
cd src                          # 进入源码目录（含有 .c 文件）
ls | grep \.c                   # 应列出 main.c, builtin.c, exec.c 等 .c 文件
ls -l | grep \.c | wc -l        # 输出 .c 文件数量（为8）
cd ..                           # 返回项目根目录
echo "hello world" | grep hello # 应输出 "hello world"
echo pipe | cat >> test/c.txt   # 追加内容
cat test/c.txt                  # 应输出 pipe

# 后台 & jobs
sleep 3 &                      # [1] PID，3秒后 Done
sleep 2 &                      # [2] PID
ls &                           # 后台列表
echo bg > test/d.txt &         # 后台重定向
cat test/d.txt                 # bg
ls | grep \.c &                # 后台管道
jobs                           # 显示作业状态

# cd / pwd
cd src
pwd                            # /src
cd -
pwd                            # 返回原目录
cd
pwd                            # /oscd
cd nonexist                    # 错误
cd -                           # 返回项目根目录并打印路径

# type
type hello                     # shell builtin
type ls                        # /bin/ls
type nonexist                  # not found
type                           # 提示用法

# 别名
alias ll='ls -l'
ll                             # ls -l
unalias ll
ll                             # 未找到

# export
export MYVAR=hello
printenv MYVAR                 # hello
export MYVAR=world
printenv MYVAR                 # world
export EMPTYVAR
printenv EMPTYVAR              # 空行
export A=1 B=2 C=3
printenv A                     # 1
export                         # 提示用法

# 变量展开
echo $HOME                     # 输出当前用户家目录（为 /home/oscd）
echo ${USER}                   # 输出当前用户名（为oscd）
echo $$                        # 输出 Shell 进程的 PID
echo $?                        # 上条命令退出码，如0
echo $NONEXIST                 # 空行（变量不存在）
echo $SHELL                    # 输出 Shell 的路径（为/bin/bash）
echo $PID                      # 输出 Shell 进程的 PID
echo $PPID                     # 输出 父进程的 PID
echo $TERM                     # 输出终端类型（为xterm-256color）
echo $SHLVL                    # 输出 Shell 嵌套层级，如 2

# 逻辑运算符 && / ||
true && echo success           # success
false && echo no               # 无输出
false || echo ok               # ok
true || echo no                # 无输出
cd . && echo ok                # ok
cd /nonexist || echo rescued   # 先报错不存在，后rescued
echo $?                        # 0（上条成功）
false                          # 无输出
echo $?                        # 1

# --help
cd --help
type --help
history --help
alias --help
pwd --help
export --help
jobs --help
true --help
false --help

# 错误处理
| ls                           # 管道符前缺少命令
ls |                           # 管道符后缺少命令
ls | | wc                      # 连续管道符
cd /tmp | ls                   # 管道中不能使用内置命令
cd /tmp &                      # 内置命令不能后台运行
ls & &                         # 多余的 &

# 历史
history -c                     # 清空
echo first
echo second
history                        # 1 echo first, 2 echo second
!5                             # 错误
!?nonexist                     # 错误
!!                             # 执行 echo second
!1                             # 执行 echo first
!?first                        # 执行 echo first
!?second? # comment            # 执行 echo second
history                        # 6条记录
history -c                     # 清空
history                        # 无输出

# 清理
clear
rm -rf test
exit
