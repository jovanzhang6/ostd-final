# 测试前执行：mkdir test

# ========== 基础管道测试 ==========
ls | grep \.c                            # 应只列出 .c 文件（如 exec.c, builtin.c, main.c）
echo "hello world" | grep world           # 应输出 hello world

# ========== 多级管道 ==========
ls -l | grep \.c | wc -l                 # 应输出 .c 文件数量（如 3）

# ========== 管道与输出重定向 ==========
ls | grep \.c > test/pipe_out.txt
cat test/pipe_out.txt                     # 应列出 .c 文件

# ========== 管道与输入重定向 ==========
echo "hello world" > test/in.txt
cat < test/in.txt | grep world           # 应输出 hello world

# ========== 管道与追加重定向 ==========
echo "first" | cat >> test/append.txt
echo "second" | cat >> test/append.txt
cat test/append.txt                       # 应输出 first\nsecond

# ========== 管道与内置命令冲突 ==========
cd /tmp | ls                              # 应报错：管道中不能使用内置命令

# ========== 语法错误测试 ==========
| ls                                      # 应报错：管道符前缺少命令
ls |                                      # 应报错：管道符后缺少命令
ls | | wc                                 # 应报错：连续管道符或空命令

# ========== 后台运行（暂不支持） ==========
ls &                                      # 应报错：后台运行尚未实现

# ========== 清理 ==========
rm test/pipe_out.txt test/in.txt test/append.txt
exit
