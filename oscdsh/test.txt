# 这是一条注释

cat test/a.txt                  # 应输出 hello

cat test/b.txt                  # 应输出 test

> test/empty.txt                # 创建空文件
ls -l test/empty.txt            # 大小应为 0

echo line2 >> test/d.txt
cat test/d.txt                  # 两行 line1 和 line2
