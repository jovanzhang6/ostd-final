# ========== 基础输入重定向 ==========
echo hello > test/in.txt
cat < test/in.txt                 # 应输出 hello

wc -c < test/in.txt               # 应输出文件字节数（含换行）

# ========== 同时使用 < 和 > ==========
echo "original" > test/source.txt
cat < test/source.txt > test/dest.txt
cat test/dest.txt                 # 应输出 original

# ========== 输入重定向 + 追加重定向 ==========
echo line1 > test/base.txt
echo line2 > test/extra.txt
cat < test/extra.txt >> test/base.txt
cat test/base.txt                 # 应显示 line1 和 line2

# ========== 错误处理 ==========
cat < nonexistent_file            # 应显示 open 错误 (perror)
<                                 # 语法错误: 缺少输入文件名
cat <                              # 同上

# ========== 清理与验证 ==========
rm test/in.txt test/source.txt test/dest.txt test/base.txt test/extra.txt
exit
