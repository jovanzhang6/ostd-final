# ==================== 准备工作 ====================
mkdir test

# ==================== 基础重定向 ====================
echo hello > test/a.txt
cat test/a.txt                 # hello
echo world >> test/a.txt
cat test/a.txt                 # hello 换行 world
wc -c < test/a.txt             # 字节数（12 左右）
cat < test/a.txt > test/b.txt
cat test/b.txt                 # 同 a.txt
> test/empty.txt
ls -l test/empty.txt           # 大小 0

# ==================== 管道 ====================
ls | grep \.c                  # 列出当前目录 .c 文件
ls -l | grep \.c | wc -l       # .c 文件数量
echo "pipe" | cat >> test/c.txt
cat test/c.txt                 # pipe

# ==================== 后台运行 ====================
sleep 2 &                      # 立即返回提示符，稍后输出 [n] Done
ls &                           # 后台列出文件（输出可能交织）
echo bg > test/d.txt &         # 后台重定向
cat test/d.txt                 # bg
ls | grep \.c &                # 后台管道

# ==================== 内置命令 cd ====================
cd /tmp
pwd                            # 期望 /tmp
cd -
pwd                            # 返回原目录
cd
pwd                            # 切换到 HOME
cd nonexist                    # 错误：oscdsh: cd: … No such file or directory

# ==================== 内置命令 type ====================
type hello                     # hello is a shell builtin
type type                      # type is a shell builtin
type ls                        # ls is /bin/ls （或相似路径）
type nonexist                  # nonexist: not found
type /bin/ls                   # /bin/ls is /bin/ls
type                           # 提示用法: type: 用法: type <命令>

# ==================== 历史与快捷操作 ====================
echo first
echo second
history                        # 应显示 1 echo first, 2 echo second
!5                             # 错误：oscdsh: !5: 无此历史编号
!?nonexist                     # 错误：oscdsh: !?nonexist?: 无匹配历史
!!                             # 应打印并执行 echo second，记录为 3
!1                             # 应打印并执行 echo first，记录为 4
!?first                        # 应打印并执行 echo first（最近匹配），记录为 5
!! #                           # 应执行上一条（echo first），记录为 6
!?second? # comment            # 应执行最近包含 second 的命令（echo second），记录为 7
history
# 最终预期历史：
#     1  echo first
#     2  echo second
#     3  echo second
#     4  echo first
#     5  echo first
#     6  echo first
#     7  echo second

# ==================== 别名 alias / unalias ====================
alias                          # 无输出（或显示无别名）
alias ll='ls -l'               # 定义别名
alias                          # 应显示 ll='ls -l'
ll                             # 应执行 ls -l
alias rm='rm -i'               # 再定义
alias rm                       # 应显示 rm='rm -i'
unalias ll                     # 删除 ll
ll                             # 提示命令未找到
unalias nonexist               # 提示未找到
alias                          # 应只剩 rm='rm -i'
alias rm='ls'                  # 重定义
alias rm                       # 应显示 rm='ls'
rm                             # 应执行 ls 而非 rm

# ==================== 内置命令 pwd / export ====================
pwd                            # 输出当前目录
export MYVAR=hello
printenv MYVAR                 # 输出 hello
export MYVAR=world
printenv MYVAR                 # 输出 world
export EMPTYVAR
printenv EMPTYVAR              # 输出空行（或 EMPTYVAR=）
export A=1 B=2 C=3
printenv A                     # 1
printenv B                     # 2
printenv C                     # 3
export                         # 提示用法

# ==================== jobs 测试 ====================
sleep 30 &                    # 应显示 [1] PID，作业 1 Running
sleep 5 &                     # 应显示 [2] PID，作业 2 Running
jobs                          # 应显示两个作业，均为 Running
# 等待 5 秒（或执行其他命令直到 sleep 5 完成），然后再次运行
jobs                          # 应显示 [1] Running，[2] Done
# 也可继续测试后台管道作业
ls | grep \.c &               # 后台管道，显示 [3] PID，状态 Running
jobs                          # 应包含新作业

# ==================== Tab 补全测试 ====================
# 请在交互环境中手动测试以下行为：
# 输入 ec 按 Tab → 应补全为 echo（若存在）
# 输入 l 按两次 Tab → 列出所有以 l 开头的命令（如 ls, less...）
# 输入 ls /tm 按 Tab → 应补全为 /tmp
# 补全后命令正常执行

# ==================== 语法错误处理 ====================
| ls                           # 语法错误：管道符前缺少命令
ls |                           # 语法错误：管道符后缺少命令
ls | | wc                      # 语法错误：连续管道符或空命令
cd /tmp | ls                   # 管道中不能使用内置命令
cd /tmp &                      # 内置命令不能后台运行
ls & &                         # 语法错误：多余的 &

# ==================== 清理 ====================
rm -rf test
exit
